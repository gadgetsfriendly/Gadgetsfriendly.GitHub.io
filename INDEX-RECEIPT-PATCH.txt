function getPendingCustomerReceipt(reference,purchaseId){
  let r=null;
  try{r=JSON.parse(localStorage.getItem('gf_pending_receipt_v1')||'null')}catch(_){}
  if(!r || r.purchaseId!==purchaseId) return null;
  r.reference=reference||r.reference||'';
  r.paymentStatus='PAID';
  try{localStorage.setItem('gf_pending_receipt_v1',JSON.stringify(r))}catch(_){}
  return r;
}

function printCustomerReceipt(){
  const r=window.__gfCustomerReceipt;
  if(!r)return alert('Receipt information is not available.');
  const w=window.open('about:blank','_blank');
  if(!w)return alert('Please allow pop-ups to open the receipt.');
  w.document.open();
  w.document.write(customerReceiptDocument(r));
  w.document.close();
}
